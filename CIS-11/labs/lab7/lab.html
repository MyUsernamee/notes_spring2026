4.ORIG X3000

6MAIN
7 LEA R6, STACK
8 STR R7, R6, #0
9 ADD R6, R6, #1
10 JSR TEST_MULT
11 JSR TEST_DIV
12 HALT
13
14TEST_MULT
15 STR R7, R6, #0
16 ADD R6, R6, #1 ; NEW STACK FRAME.
17 LD R1, X
18 LD R2, Y
19
20 JSR MULT
21 LD R2, MULT_TEST_EXPECTED
22 AND R1, R1, #0
23 ADD R1, R1, R0
24 JSR ASSERT_EQ
25
26 ADD R6, R6, #-1
27 LDR R7, R6, #-1
28 RET
29
30TEST_DIV
31 STR R7, R6, #0
32 ADD R6, R6, #1 ; NEW STACK FRAME.
33
34 AND R1, R1, #0
35 AND R2, R2, #0 ; Clear R1, R2
36
37 ADD R1, R1, #15
38 ADD R2, R2, #3 ; R1 = 15, R2 = 3
39
40 JSR DIV ; R0 = R1 / R2
41
42 AND R1, R1, #0
43 ADD R1, R1, R0 ; R1 = R0
44
45 AND R2, R2, #0
46 ADD R2, R2, #5 ; R2 = #5
47
48 JSR ASSERT_EQ ; assert(R1 == 5)
49
50 ADD R6, R6, #-1
51 LDR R7, R6, #-1
52 RET
53
54ASSERT_EQ ; IF R1==R2, HALTS, AND PRINTS OUTPUT IF NOT.
55 STR R7, R6, #0
56 ADD R6, R6, #1 ; NEW STACK FRAME.
57 NOT R1, R1
58 ADD R1, R1, #1
59
60 ADD R2, R2, R1
61
62 BRZ SUCCESS
63
64 LEA R0, ASSERT_MESSAGE
65 PUTS
66 HALT
67
68SUCCESS
69 ADD R6, R6, #-1
70 LDR R7, R6, #-1
71 RET
72
73; END ASSERT_EQ
74
75MULT ; MULTIPLIES TWO NUMBERS
76 STR R7, R6, #0
77 ADD R6, R6, #1 ; NEW STACK FRAME.
78
79 AND R0, R0, #0 ; CLEAR R0
80
81LOOP_MULT
82 ADD R1, R1, #-1
83 BRN DONE
84 ADD R0, R0, R2
85 BR LOOP_MULT
86
87; end MULT
88
89DIV ; DIVIDES TO INTERGERS
90 STR R7, R6, #0
91
92 ADD R6, R6, #1 ; NEW STACK FRAME.
93 AND R0, R0, #0 ; CLEAR R0
94
95 NOT R2, R2
96 ADD R2, R2, #1 ; Invert R2
97
98LOOP_DIV
99 ADD R0, R0, #1
100 ADD R1, R1, R2
101 BRZ DONE
102 BRN DONE
103 BR LOOP_DIV
104
105DONE
106 ADD R6, R6, #-1
107 LDR R7, R6, #-1
108 RET
109
110; end MULT
